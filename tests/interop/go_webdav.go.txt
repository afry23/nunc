// Interop-Helfer für `webdav-go-nunc.sh`: ein WebDAV-Server und ein Sync-Lauf mit der echten Logik der Go-App.
// Endung .txt, damit die Datei hier nicht als Go-Code gilt.
//
//	go_webdav serve <ordner> <port> <benutzer> <passwort>
//	go_webdav sync <datenbank> <url> <benutzer> <passwort> <geräte-id> <remote-pfad>
package main

import (
	"context"
	"fmt"
	"net/http"
	"os"

	"golang.org/x/net/webdav"

	"nunc/internal/db"
	"nunc/internal/models"
	"nunc/internal/services"
)

func must(err error) {
	if err != nil {
		fmt.Fprintln(os.Stderr, "FEHLER:", err)
		os.Exit(1)
	}
}

func main() {
	switch os.Args[1] {
	case "serve":
		handler := &webdav.Handler{FileSystem: webdav.Dir(os.Args[2]), LockSystem: webdav.NewMemLS()}
		user, password := os.Args[4], os.Args[5]
		must(http.ListenAndServe("127.0.0.1:"+os.Args[3], http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			u, p, ok := r.BasicAuth()
			if !ok || u != user || p != password {
				w.Header().Set("WWW-Authenticate", `Basic realm="nunc-interop"`)
				http.Error(w, "unauthorized", http.StatusUnauthorized)
				return
			}
			handler.ServeHTTP(w, r)
		})))
	case "sync":
		d, err := db.OpenDB(os.Args[2])
		must(err)
		cfg := models.WebDAVConfig{URL: os.Args[3], AuthMethod: "basic", Username: os.Args[4], RemotePath: os.Args[7]}
		svc := services.NewWebDAVSyncService(cfg, os.Args[5], d, nil, os.Args[6])
		must(svc.Sync(context.Background()))
		fmt.Println("go sync ok")
	}
}
