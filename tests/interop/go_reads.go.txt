// Interop-Check: liest eine von nunc (Rust) geschriebene Datenbank mit der Logik der Go-App.
// Aufruf siehe go-reads-nunc-db.sh; Endung .txt, damit die Datei hier nicht als Go-Code gilt.
package main

import (
	"encoding/json"
	"fmt"
	"os"

	"nunc/internal/db"
	"nunc/internal/models"
	"nunc/internal/services"
)

func must(err error) {
	if err != nil {
		fmt.Fprintln(os.Stderr, "FEHLER:", err)
		os.Exit(1)
	}
}

func main() {
	d, err := db.OpenDB(os.Args[1])
	must(err)

	var integrity string
	must(d.QueryRow(`PRAGMA integrity_check`).Scan(&integrity))

	settings, err := services.NewSettingsService(d)
	must(err)
	todos, err := services.NewTodoService(d).GetAll()
	must(err)
	stats, err := services.NewStatsService(d).GetStats(models.StatsRequest{
		Period: "month", StartDate: "2026-10-01", EndDate: "2026-10-31",
	})
	must(err)

	var sessions, versions int
	must(d.QueryRow(`SELECT COUNT(*) FROM sessions`).Scan(&sessions))
	must(d.QueryRow(`SELECT COUNT(*) FROM schema_migrations`).Scan(&versions))

	// das von nunc geschriebene Backup muss die Restore-Prüfung der Go-App bestehen
	backupCheck := "ok"
	if len(os.Args) > 2 {
		if err := services.ValidateNuncBackupFile(os.Args[2]); err != nil {
			backupCheck = err.Error()
		}
	}

	out := map[string]any{
		"backupCheck":    backupCheck,
		"integrity":      integrity,
		"settings":       settings.Get(),
		"todos":          todos,
		"totalPomodoros": stats.TotalPomodoros,
		"totalWorkMin":   stats.TotalWorkMin,
		"sessions":       sessions,
		"migrations":     versions,
	}
	enc := json.NewEncoder(os.Stdout)
	enc.SetIndent("", "  ")
	must(enc.Encode(out))
}
